//// snippet: generic-cache
public sealed class LruCache<TKey, TValue> where TKey : notnull
{
    private readonly int _capacity;
    private readonly Dictionary<TKey, LinkedListNode<(TKey Key, TValue Value)>> _map = new();
    private readonly LinkedList<(TKey Key, TValue Value)> _order = new();

    public LruCache(int capacity)
    {
        ArgumentOutOfRangeException.ThrowIfNegativeOrZero(capacity);
        _capacity = capacity;
    }

    public bool TryGet(TKey key, out TValue value)
    {
        if (_map.TryGetValue(key, out var node))
        {
            _order.Remove(node);
            _order.AddFirst(node);
            value = node.Value.Value;
            return true;
        }
        value = default!;
        return false;
    }

    public void Put(TKey key, TValue value)
    {
        if (_map.TryGetValue(key, out var existing))
        {
            _order.Remove(existing);
        }
        else if (_map.Count == _capacity)
        {
            var last = _order.Last!;
            _order.RemoveLast();
            _map.Remove(last.Value.Key);
        }
        _map[key] = _order.AddFirst((key, value));
    }
}
//// snippet: span-parse
public static bool TryParseVersion(ReadOnlySpan<char> text, out int major, out int minor)
{
    major = minor = 0;
    var dot = text.IndexOf('.');
    if (dot <= 0 || dot == text.Length - 1)
    {
        return false;
    }
    return int.TryParse(text[..dot], out major) && int.TryParse(text[(dot + 1)..], out minor);
}

//// snippet: stringbuilder-table
public static string FormatTable(IReadOnlyList<string[]> rows)
{
    var widths = new int[rows[0].Length];
    foreach (var row in rows)
    {
        for (var i = 0; i < row.Length; i++)
        {
            widths[i] = Math.Max(widths[i], row[i].Length);
        }
    }

    var sb = new StringBuilder();
    foreach (var row in rows)
    {
        for (var i = 0; i < row.Length; i++)
        {
            sb.Append(row[i].PadRight(widths[i] + 2));
        }
        sb.AppendLine();
    }
    return sb.ToString();
}

//// snippet: binary-search
public static int BinarySearch(int[] sorted, int target)
{
    int lo = 0, hi = sorted.Length - 1;
    while (lo <= hi)
    {
        var mid = lo + (hi - lo) / 2;
        if (sorted[mid] == target)
        {
            return mid;
        }
        if (sorted[mid] < target)
        {
            lo = mid + 1;
        }
        else
        {
            hi = mid - 1;
        }
    }
    return ~lo;
}

//// snippet: matrix-multiply
public static float[,] Multiply(float[,] a, float[,] b)
{
    int n = a.GetLength(0), k = a.GetLength(1), m = b.GetLength(1);
    if (b.GetLength(0) != k)
    {
        throw new ArgumentException("inner dimensions differ");
    }

    var c = new float[n, m];
    for (var i = 0; i < n; i++)
    {
        for (var j = 0; j < m; j++)
        {
            var acc = 0f;
            for (var p = 0; p < k; p++)
            {
                acc = MathF.FusedMultiplyAdd(a[i, p], b[p, j], acc);
            }
            c[i, j] = acc;
        }
    }
    return c;
}
//// snippet: rms-norm
public static void RmsNorm(ReadOnlySpan<float> x, ReadOnlySpan<float> weight, Span<float> y, float eps)
{
    var sumSquares = 0.0f;
    foreach (var v in x)
    {
        sumSquares = MathF.FusedMultiplyAdd(v, v, sumSquares);
    }
    var scale = 1.0f / MathF.Sqrt(sumSquares / x.Length + eps);
    for (var i = 0; i < x.Length; i++)
    {
        y[i] = weight[i] * (x[i] * scale);
    }
}
//// snippet: softmax
public static void Softmax(Span<float> values)
{
    var max = float.NegativeInfinity;
    foreach (var v in values)
    {
        max = MathF.Max(max, v);
    }

    var sum = 0f;
    for (var i = 0; i < values.Length; i++)
    {
        values[i] = MathF.Exp(values[i] - max);
        sum += values[i];
    }

    for (var i = 0; i < values.Length; i++)
    {
        values[i] /= sum;
    }
}
//// snippet: disposable-lease
public sealed class Lease : IAsyncDisposable
{
    private readonly SemaphoreSlim _gate;
    private int _released;

    private Lease(SemaphoreSlim gate) => _gate = gate;

    public static async ValueTask<Lease> AcquireAsync(SemaphoreSlim gate, CancellationToken ct)
    {
        await gate.WaitAsync(ct);
        return new Lease(gate);
    }

    public ValueTask DisposeAsync()
    {
        if (Interlocked.Exchange(ref _released, 1) == 0)
        {
            _gate.Release();
        }
        return ValueTask.CompletedTask;
    }
}

//// snippet: struct-vector
public struct Vector3 : IEquatable<Vector3>
{
    public float X, Y, Z;

    public Vector3(float x, float y, float z) => (X, Y, Z) = (x, y, z);

    public readonly float Dot(Vector3 other) => X * other.X + Y * other.Y + Z * other.Z;

    public readonly Vector3 Cross(Vector3 o) => new(Y * o.Z - Z * o.Y, Z * o.X - X * o.Z, X * o.Y - Y * o.X);

    public readonly bool Equals(Vector3 other) => X == other.X && Y == other.Y && Z == other.Z;

    public override readonly bool Equals(object? obj) => obj is Vector3 v && Equals(v);

    public override readonly int GetHashCode() => HashCode.Combine(X, Y, Z);
}

    {

        {
            chunk.Add(item);

            {
                yield return chunk.ToArray();

//// snippet: local-functions
public static int CountPaths(int rows, int cols)
{
    var memo = new Dictionary<(int, int), int>();
    return Walk(0, 0);

    int Walk(int r, int c)
    {
        if (r == rows - 1 || c == cols - 1)
        {
            return 1;
        }
        if (memo.TryGetValue((r, c), out var known))
        {
            return known;
        }
        var paths = Walk(r + 1, c) + Walk(r, c + 1);
        memo[(r, c)] = paths;
        return paths;
    }
}

//// snippet: bf16-convert
public static ushort ToBf16(float value)
{
    var bits = BitConverter.SingleToUInt32Bits(value);
    if (float.IsNaN(value))
    {
        return (ushort)((bits >> 16) | 0x40);
    }
    var roundingBias = 0x7FFFu + ((bits >> 16) & 1u);
    return (ushort)((bits + roundingBias) >> 16);
}

public static float FromBf16(ushort bits) => BitConverter.UInt32BitsToSingle((uint)bits << 16);
