public static class MathUtil
{
    public static int Add(int a, int b) => a + b;

    public static long Factorial(int n)
    {
        if (n < 0) throw new ArgumentOutOfRangeException(nameof(n));
        long acc = 1L;
        for (var i = 2; i <= n; i++) acc *= i;
        return acc;
    }
}

public async Task<string> FetchAsync(HttpClient client, Uri uri, CancellationToken ct = default)
{
    using var response = await client.GetAsync(uri, ct).ConfigureAwait(false);
    response.EnsureSuccessStatusCode();
    return await response.Content.ReadAsStringAsync(ct);
}

static string Describe(object? o) => o switch
{
    null => "null",
    int i when i < 0 => "negative int",
    int i => $"int {i}",
    string { Length: 0 } => "empty string",
    string s => $"string of {s.Length}",
    IEnumerable<int> xs => $"ints, sum {xs.Sum()}",
    _ => o.GetType().Name,
};

/// <summary>
/// Returns the index of <paramref name="value"/> in <paramref name="span"/>, or -1.
/// </summary>
/// <typeparam name="T">The element type; must implement <see cref="IEquatable{T}"/>.</typeparam>
public static int IndexOf<T>(ReadOnlySpan<T> span, T value) where T : IEquatable<T>
{
    for (int i = 0; i < span.Length; i++)
    {
        if (span[i].Equals(value)) return i;
    }
    return -1;
}

public class Matrix
{
    private readonly float[,] _cells;
    public Matrix(int rows, int cols) => _cells = new float[rows, cols];
    public float this[int r, int c] { get => _cells[r, c]; set => _cells[r, c] = value; }
    public int Rows => _cells.GetLength(0);
    public event EventHandler<int>? Changed;
    protected virtual void OnChanged(int row) => Changed?.Invoke(this, row);
    ~Matrix() { }
}

namespace Lrm.Core
{
    using static System.Math;

    internal static class Rope
    {
        // freq = f32(f32(pos) * invFreq); cos = f32(cos(f64 freq))
        public static float Cos(int pos, float invFreq) => (float)Cos((double)((float)pos * invFreq));
    }
}
